    SCHEMES {
        varchar id PK "Strata Plan ID e.g. SP52042"
        varchar name "Scheme Name"
        text address "Full Street Address"
        int lots "Total Lot Count"
        boolean active "Active Flag"
        timestamptz created_at "Created Timestamp"
    }

    UNITS {
        uuid id PK "Unique Unit Record ID"
        varchar scheme_id FK "References SCHEMES"
        varchar unit_id "Unit Label e.g. Unit 4"
        int lot_number "Lot Number e.g. 4"
        numeric entitlement "Statutory Unit Entitlement Weight"
        varchar status "Occupied or Vacant"
        timestamptz created_at "Created Timestamp"
    }

    PROFILES {
        uuid id PK "References auth.users"
        varchar full_name "Full Name"
        varchar email "Unique Email Address"
        varchar phone_number "Phone Number"
        text avatar_url "Profile Avatar URL"
        boolean is_system_admin "Superadmin Bypass Flag"
        timestamptz created_at "Created Timestamp"
    }

    MEMBERS {
        uuid id PK "Unique Member Record"
        varchar scheme_id FK "References SCHEMES"
        uuid user_id FK "References PROFILES nullable on invite"
        varchar name "Display Name"
        varchar email "Contact Email"
        varchar phone "Contact Phone"
        varchar role "Strata Role"
        varchar committee_office "Office: Chairperson, Treasurer, Secretary"
        varchar unit_id FK "References UNITS"
        int lot_number "Associated Lot Number"
        boolean is_co_owner "Co-Owner or Partner Flag"
        uuid primary_member_id FK "Self References MEMBERS"
        varchar status "Active, Invited, Pending Verification, Restricted"
        date joined_at "Date Joined Scheme"
        timestamptz created_at "Created Timestamp"
    }

    MEMBER_ONBOARDING_VERIFICATIONS {
        uuid id PK "Verification Record"
        uuid member_id FK "References MEMBERS Unique"
        varchar id_document_number "Government ID Number"
        text id_doc_url "Uploaded ID Document URL"
        text ownership_doc_url "Title Deed or Tenancy Agreement URL"
        boolean two_factor_verified "2FA Confirmation Status"
        uuid verified_by FK "References PROFILES"
        timestamptz verified_at "Verification Timestamp"
        timestamptz created_at "Created Timestamp"
    }

    RESIDENT_REQUESTS {
        uuid id PK "Primary Key"
        varchar reference_id "Unique Ref e.g. SL-10452"
        varchar scheme_id FK "References SCHEMES"
        varchar unit_id "Optional Unit Context"
        varchar title "Request Title"
        text description "Detailed Description"
        varchar request_type "Maintenance, Emergency, Complaint, Unit, Recurring"
        varchar priority "Low, Normal, Medium, High, Urgent, Emergency"
        varchar location "Lobby, Lift, Common Area, Roof, etc."
        varchar contact_preference "Email or SmartLot Notifications"
        date due_date "Optional Due Date"
        varchar status "new, needs_triage, in_voting, approved, rejected, closed"
        uuid requestor_id FK "References PROFILES"
        text rejection_reason "Reason if Rejected at Triage"
        text close_reason "Reason if Closed Early"
        uuid closed_by FK "References PROFILES"
        timestamptz closed_at "Closed Timestamp"
        uuid assigned_to_id FK "References PROFILES"
        timestamptz created_at "Created Timestamp"
        timestamptz updated_at "Updated Timestamp"
    }

    ACTIVITY_ATTACHMENTS {
        uuid id PK "Attachment Record"
        uuid request_id FK "References RESIDENT_REQUESTS"

        bigint file_size_bytes "Size in Bytes"

        varchar version_type "original or revised Miro Resubmission"
        text note "Annotation or Revision Note"
        uuid uploaded_by FK "References PROFILES"
        timestamptz created_at "Created Timestamp"
    }

    REQUEST_COMMENTS {
        uuid id PK "Comment Record"
        uuid request_id FK "References RESIDENT_REQUESTS"

        uuid reply_to_id FK "References REQUEST_COMMENTS"
        boolean is_email_reply "True if received via inbound email"

    ACTIVITY_NOTES {
        uuid id PK "Internal Note Record"
        uuid request_id FK "References RESIDENT_REQUESTS"

    }

    ACTIVITY_AUDIT_LOG {
        uuid id PK "Audit Event"
        uuid request_id FK "References RESIDENT_REQUESTS"
        uuid actor_id FK "References PROFILES nullable for system"

        text note "Event Description"
        varchar from_status "Previous Status"
        varchar to_status "New Status"
        timestamptz created_at "Event Timestamp"
    }

    MOTIONS {
        uuid id PK "Motion Record"
        uuid request_id FK "References RESIDENT_REQUESTS"
        varchar scheme_id FK "References SCHEMES"
        varchar motion_reference "Unique Ref e.g. MOT-SP52042-014"
        varchar title "SP - Address - Heading"
        varchar heading "Short Heading"
        text summary "Legal or Statutory Motion Summary"
        varchar voter_group "committee_only, lot_owners, all_residents"
        varchar result_criteria "simple_majority, special_resolution, unanimous"
        int committee_size "Total Eligible Voters e.g. 6"
        int quorum_target "Votes Required to Pass e.g. 4"
        timestamptz deadline "Voting Cutoff Timestamp"
        timestamptz original_deadline "Initial Cutoff Before RFI"
        varchar status "active, passed, rejected, unresolved, closed_early"
        text close_reason "Reason if Closed Early by Manager"
        uuid closed_by FK "References PROFILES"
        timestamptz closed_at "Closed Timestamp"
        uuid created_by FK "References PROFILES"
        timestamptz created_at "Created Timestamp"
        timestamptz updated_at "Updated Timestamp"
    }

    MOTION_BALLOTS {
        uuid id PK "Ballot Record"
        uuid motion_id FK "References MOTIONS"
        uuid member_id FK "References MEMBERS"
        varchar vote "YES, NO, ABSTAIN"
        text comment "Voter Rationale or SCM Feedback"
        timestamptz voted_at "Timestamp Ballot Cast"
    }

    MOTION_RFIS {
        uuid id PK "RFI Record"
        uuid motion_id FK "References MOTIONS"
        uuid requested_by FK "References MEMBERS"

        text response_note "Manager or Requestor Answer"
        uuid responded_by FK "References PROFILES"
        timestamptz responded_at "Response Timestamp"
        timestamptz created_at "Created Timestamp"
    }

    MOTION_RESTARTS {
        uuid id PK "Restart Event"
